import hashlib import json import os import urllib.request # eCFR API endpoint tracking Title 15 regulatory version changes VERSION_URL = "https://www.ecfr.gov/api/versioner/v1/versions/title-15.json" STATE_FILE = "matrix-state.json" def fetch_live_version_data(): req = urllib.request.Request( VERSION_URL, headers={'User-Agent': 'JurisMatrix-AutomationBot/1.0'} ) with urllib.request.urlopen(req) as response: return json.loads(response.read().decode()) def get_current_hash(data): raw_str = json.dumps(data, sort_keys=True) return hashlib.sha256(raw_str.encode()).hexdigest() def check_and_update(): live_data = fetch_live_version_data() new_hash = get_current_hash(live_data) old_hash = "" if os.path.exists(STATE_FILE): with open(STATE_FILE, "r") as f: old_state = json.load(f) old_hash = old_state.get("hash", "") if new_hash != old_hash: print(f"Regulatory change detected! Hash updated from {old_hash} to {new_hash}") # Save new state hash with open(STATE_FILE, "w") as f: json.dump({"hash": new_hash}, f, indent=2) return True else: print("No regulatory changes detected in eCFR Title 15 source.") return False if __name__ == "__main__": changed = check_and_update() # Exit with code 0 if unchanged, or custom handling if needed exit(0 if not changed else 0)
